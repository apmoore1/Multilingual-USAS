/* new autotag rules for SEMTAG - 13 Jun 96 */
/* converted from the CLAWS C7 tagset to the UPOS tagset */

/* if first column matches original word form */
/* and second column matches original UPOS tag */
/* then tag word as directed by semantic tag list */
/* list is processed in order, so ONLY first matching entry has any effect */

/* amount of money: 54p = pence */

*[0-9]p                  NOUN    I1
*pence                   NOUN    I1

/* forties, fifties ... */

*s                       NUM     T1.3 N1 T3

/* correct CLAWS error : I sometimes tagged MC1 or ZZ1 (now NUM or SYM) */

I                        NUM     Z8
I                        SYM     Z8
i                        SYM     Z8

/* numbers with a unit suffix (million, thousand): were after the MC* catch-all in the original so never matched, moved before it */

*M                       NUM     N1
[0-9][0-9]*K             NUM     N1

/* fractions: only the numeric form, e.g. 3/4 (was * MF*) */

[0-9][0-9]*/[0-9][0-9]*  NUM     N5

/* all other MC* are numbers */

*                        NUM     N1 T1.2 T3 T1.3 N3.2

/* ordinals (was * MD* = linear order or time): restricted to digit forms, e.g. 21st */

[0-9][0-9]*st            ADJ     N4 T1.3
[0-9][0-9]*nd            ADJ     N4 T1.3
[0-9][0-9]*rd            ADJ     N4 T1.3
[0-9][0-9]*th            ADJ     N4 T1.3

/* percentages */

*%                       NOUN    N5

/* money */

£*                       NOUN    I1
$*                       NOUN    I1

/* times */

*am                      ADV     T1.2
*o'clock                 ADV     T1.2
*pm                      ADV     T1.2
[0-9][0-9]*ish           ADJ     T3 N1

/* weights and measures: a digit is required before the unit */

*[0-9]M                  NOUN    N3.3 N3.7
*[0-9]cc                 NOUN    N3.4
*[0-9]cm                 NOUN    N3.3 N3.7
*[0-9]cms                NOUN    N3.3 N3.7
*[0-9]ft                 NOUN    N3.3 N3.7
*[0-9]gr                 NOUN    N3.5
*[0-9]in                 NOUN    N3.3 N3.7
*[0-9]ins                NOUN    N3.3 N3.7
[0-9][0-9]*K             NOUN    N3.5
*[0-9]kg                 NOUN    N3.5
*[0-9]kgs                NOUN    N3.5
*[0-9]lb                 NOUN    N3.5
*[0-9]lbs                NOUN    N3.5
*[0-9]mm                 NOUN    N3.3 N3.7
*[0-9]oz                 NOUN    N3.5
*[0-9]ozs                NOUN    N3.5
*kph                     NOUN    N3.8
*[0-9]km                 NOUN    N3.3 N3.7
*[0-9]m                  NOUN    N3.3 N3.7
*[0-9]m2                 NOUN    N3.6
*[0-9]m3                 NOUN    N3.4

/* temperature */

*°C                      X       O4.6
*°F                      X       O4.6

/* times formula */

[0-9][0-9]*x[0-9][0-9]*  X       N1 N3.3

/* road numbers */

A[0-9][0-9]*             SYM     Z2
B[0-9][0-9]*             SYM     Z2
M[0-9][0-9]*             SYM     Z2

/* all other ZZs are letters (now all other symbols) */

*                        SYM     Z5
