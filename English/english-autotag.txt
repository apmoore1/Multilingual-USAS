/* new autotag rules for SEMTAG - 13 Jun 96 */

/* if first column matches original word form */
/* and second column matches original CLAWS tag */
/* then tag word as directed by semantic tag list */

/* list is processed in order, so ONLY first matching entry has any effect */

/* amount of money: 54p = pence */

*p                       NNU     I1
*pence                   NNU2    I1

/* forties, fifties ... */

*s                       MC2     T1.3 N1 T3

/* correct CLAWS error : I sometimes tagged MC1 or ZZ1 */

I                        MC1     Z8
I                        ZZ1     Z8
i                        ZZ1     Z8

/* all other MC* are numbers */

*                        MC*     N1 T1.2 T3 T1.3 N3.2

/* all MD* are linear order or time */

*                        MD*     N4 T1.3

/* percentages */

*%                       NNU     N5

/* fractions */

*                        MF*     N5

/* money */

#*                       NNU     I1
$*                       NNU     I1
&pound;*                 NNU     I1
&dollar;*                NNU     I1


/* times */

*am                      RA      T1.2
*o'clock                 RA      T1.2
*pm                      RA      T1.2
[0-9]*ish                JJ      T3 N1

/* weights and measures */

*M                       MC      N1 
*M                       NNU     N3.3 N3.7
*cc                      NNU     N3.4
*cm                      NNU     N3.3 N3.7
*cms                     NNU2    N3.3 N3.7
*ft                      NNU     N3.3 N3.7
*gr                      NNU     N3.5
*in                      NNU     N3.3 N3.7
*ins                     NNU2    N3.3 N3.7
[0-9]*K                  MC      N1 
[0-9]*K                  NNU     N3.5
*kg                      NNU     N3.5
*kgs                     NNU2    N3.5
*lb                      NNU     N3.5
*lbs                     NNU2    N3.5
*mm                      NNU     N3.3 N3.7
*oz                      NNU     N3.5
*ozs                     NNU2    N3.5
*kph                     NNU     N3.8
*km                      NNU     N3.3 N3.7
*m                       NNU     N3.3 N3.7
*m2                      NNU     N3.6
*m3                      NNU     N3.4

/* temperature */
*&deg;C                  FO      O4.6
*&deg;F                  FO      O4.6


/* times formula */

*x*                      FO      N1 N3.3
